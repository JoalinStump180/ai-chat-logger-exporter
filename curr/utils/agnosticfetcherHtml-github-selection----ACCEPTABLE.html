2. Trigger the Disk Download via Blob + URL.createObjectURL

Create a Blob, generate a temporary object URL, click a hidden anchor with download, and revoke the URL right after:

JavaScript
  function dumpSelectionFragmentToDisk(filename = "github-selection.html") {
    const frag = selectionFragmentFromLiveSelection();
    if (!frag) return false;

    const html = fragmentToHtmlString(frag);
    const blob = new Blob([html], { type: "text/html;charset=utf-8" });
    const url = URL.createObjectURL(blob);

    const a = document.createElement("a");
    a.href = url;
    a.download = filename.endsWith(".html") ? filename : `${filename}.html`;
    document.body.appendChild(a);
    a.click();
    a.remove();

    setTimeout(() => URL.revokeObjectURL(url), 1000);
    return true;
  }
